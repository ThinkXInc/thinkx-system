# 運用手順のコマンド一覧
#

# 今つながっているIPをterraformで許可リストに追加
```bash
cd ~/Sources/thinkx-system
bash infra/scripts/add_current_office_ip.sh
```

# 経路疎通チェック(Mac→LB→web→Flask をホップごとに緑/赤で確認)
```bash
cd ~/Sources/thinkx-system
bash infra/scripts/check_request_path.sh $(terraform -chdir=infra/terraform output -raw lb_public_ip) supercom-web1 supercom-lb1
```

# 受け入れ試験(3サイト全ゴールデンルートを LB へ照合)
```bash
cd ~/Sources/thinkx-system
bash infra/scripts/acceptance-sweep.sh $(terraform -chdir=infra/terraform output -raw lb_public_ip)
```

# SSH 許可 IP の棚卸し(IP ごとの最終アクセス実測・未使用は黄)
```bash
cd ~/Sources/thinkx-system
bash infra/scripts/office_ip_report.sh
```

# DNS 切替前にブラウザで本番を見る(vhost 方式のため IP 直打ちは不可・確認後は削除)
```bash
printf '52.197.179.70 thinkxinc.com transformism.art kazukiotsuka.com\n' | sudo tee -a /etc/hosts
```
```bash
sudo sed -i.bak '/^52.197.179.70 /d' /etc/hosts
```
